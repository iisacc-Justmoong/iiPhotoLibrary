cmake_minimum_required(VERSION 3.31)
project(PhotoLvuiConsumer LANGUAGES CXX)
include(CTest)
find_package(iiPhotoLibraryPresentation CONFIG REQUIRED)
find_package(LVUI 0.1 CONFIG REQUIRED COMPONENTS Native)
find_package(iiGUI 0.5 CONFIG REQUIRED)
set(IIGUI_WIDGET_STATE_HEADER "" CACHE FILEPATH "iiGUI private headless test host")
if(NOT EXISTS "${IIGUI_WIDGET_STATE_HEADER}")
    message(FATAL_ERROR "IIGUI_WIDGET_STATE_HEADER must identify the headless test host")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/private/iiGUI/WidgetState.h"
    CONTENT "#pragma once\n#include \"${IIGUI_WIDGET_STATE_HEADER}\"\n")
lvui_generate(NAME PhotosView SOURCE "${iiPhotoLibraryPresentation_VIEWS_DIR}/PhotosView.lvui"
    FRAMEWORK iiGUI INCLUDES iiPhotoLibrary/PhotoViewModel.h OUTPUT_VARIABLE generated)
add_executable(PhotoLvuiConsumer consumer.cpp mount.cpp "${generated}")
target_compile_features(PhotoLvuiConsumer PRIVATE cxx_std_23)
target_include_directories(PhotoLvuiConsumer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/private")
target_link_libraries(PhotoLvuiConsumer PRIVATE iiPhotoLibrary::Presentation LVUI::Native iiGUI::iiGUI)
add_test(NAME iiPhotoLibrary.LvuiConsumer COMMAND PhotoLvuiConsumer)
set_tests_properties(iiPhotoLibrary.LvuiConsumer PROPERTIES TIMEOUT 30)
if(APPLE)
    set_tests_properties(iiPhotoLibrary.LvuiConsumer PROPERTIES ENVIRONMENT_MODIFICATION
        "DYLD_LIBRARY_PATH=unset:;DYLD_FALLBACK_LIBRARY_PATH=unset:;DYLD_FRAMEWORK_PATH=unset:;DYLD_FALLBACK_FRAMEWORK_PATH=unset:")
endif()
