cmake_minimum_required(VERSION 3.24)
project(PhotoPresentationBridgeConsumer LANGUAGES CXX)
include(CTest)
find_package(iiPhotoLibraryPresentationBridge CONFIG REQUIRED)
find_package(Qt6 6.8.3 EXACT COMPONENTS Test REQUIRED)
add_executable(PhotoPresentationBridgeConsumer ../photo_presentation_bridge.cpp)
target_compile_features(PhotoPresentationBridgeConsumer PRIVATE cxx_std_23)
target_link_libraries(PhotoPresentationBridgeConsumer PRIVATE iiPhotoLibrary::PresentationBridge Qt6::Test)
set_target_properties(PhotoPresentationBridgeConsumer PROPERTIES AUTOMOC ON)
add_test(NAME iiPhotoLibrary.InstalledPresentationBridge COMMAND PhotoPresentationBridgeConsumer)
