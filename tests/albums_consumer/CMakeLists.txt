cmake_minimum_required(VERSION 3.24)
project(AlbumsConsumer LANGUAGES CXX)
find_package(iiPhotoLibrary 0.1 CONFIG REQUIRED COMPONENTS Albums)
if(TARGET Qt6::Core OR TARGET iiPhotoLibrary::iiPhotoLibrary)
    message(FATAL_ERROR "Albums unexpectedly loaded the legacy photo/UI layer")
endif()
add_executable(AlbumsConsumer main.cpp)
target_link_libraries(AlbumsConsumer PRIVATE iiPhotoLibrary::Albums)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/run-path.txt" CONTENT "$<TARGET_FILE:AlbumsConsumer>")
